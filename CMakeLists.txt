cmake_minimum_required(VERSION 3.16)
project(receiver VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This firmware uses Linux DVB, V4L2 and POSIX APIs. Build on Linux or with an AArch64 Linux toolchain.")
endif()
find_package(Threads REQUIRED)
add_executable(receiver firmware/c/main.c firmware/c/config.c firmware/c/radio.c firmware/c/television.c firmware/c/ethernet.c)
target_compile_definitions(receiver PRIVATE _DEFAULT_SOURCE)
target_compile_options(receiver PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(receiver PRIVATE Threads::Threads m)
install(TARGETS receiver RUNTIME DESTINATION bin)
